cmake_minimum_required(VERSION 3.25)
project(slopcap_camprobe CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

add_executable(slopcap-camprobe main.cpp report.cpp)
target_compile_options(slopcap-camprobe PRIVATE -Wall -Wextra)

if(WIN32)
  # Cross build (mingw-w64.cmake): link the vendor DLLs copied from the target machine; use the Linux packages'
  # SDK headers through a private include folder so /usr/include itself never enters a Windows compile.
  set(WIN_SDK_DLL_DIR "" CACHE PATH "folder with ASICamera2.dll and qhyccd.dll from the target machine")
  if(NOT WIN_SDK_DLL_DIR)
    message(FATAL_ERROR "set -DWIN_SDK_DLL_DIR=<folder with ASICamera2.dll and qhyccd.dll>")
  endif()
  set(SDK_INC ${CMAKE_CURRENT_BINARY_DIR}/sdk-include)
  file(MAKE_DIRECTORY ${SDK_INC})
  file(CREATE_LINK /usr/include/libasi ${SDK_INC}/libasi SYMBOLIC)
  target_include_directories(slopcap-camprobe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/compat/qhy ${SDK_INC} /usr/include/libqhy)
  target_sources(slopcap-camprobe PRIVATE probe_asi.cpp probe_qhy.cpp)
  target_link_libraries(slopcap-camprobe PRIVATE ${WIN_SDK_DLL_DIR}/ASICamera2.dll ${WIN_SDK_DLL_DIR}/qhyccd.dll)
  target_compile_definitions(slopcap-camprobe PRIVATE CAMPROBE_ASI CAMPROBE_QHY)
  return()
endif()

if(APPLE)
  # The dylibs the top-level CMakeLists.txt copied out of SLOPCAP_MAC_SDK_DIR (an indi-3rdparty checkout).
  foreach(lib ${SLOPCAP_MAC_SDK_DYLIBS})
    target_link_libraries(slopcap-camprobe PRIVATE ${lib})
    if(lib MATCHES "libASICamera2")
      target_sources(slopcap-camprobe PRIVATE probe_asi.cpp)
      target_include_directories(slopcap-camprobe PRIVATE ${SLOPCAP_MAC_SDK_DIR})
      target_compile_definitions(slopcap-camprobe PRIVATE CAMPROBE_ASI)
    elseif(lib MATCHES "libqhyccd")
      target_sources(slopcap-camprobe PRIVATE probe_qhy.cpp)
      target_include_directories(slopcap-camprobe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/compat/qhy ${SLOPCAP_MAC_SDK_DIR}/libqhy)
      target_compile_definitions(slopcap-camprobe PRIVATE CAMPROBE_QHY)
    endif()
  endforeach()
  if(NOT SLOPCAP_MAC_SDK_DYLIBS)
    message(STATUS "camprobe: no camera SDK (set SLOPCAP_MAC_SDK_DIR to an indi-3rdparty checkout)")
  endif()
  return()
endif()

# ZWO ASI SDK (Debian/Ubuntu package layout: <libasi/ASICamera2.h>, libASICamera2.so)
find_path(ASI_INCLUDE_DIR libasi/ASICamera2.h)
find_library(ASI_LIBRARY ASICamera2)
if(ASI_INCLUDE_DIR AND ASI_LIBRARY)
  target_sources(slopcap-camprobe PRIVATE probe_asi.cpp)
  target_include_directories(slopcap-camprobe PRIVATE ${ASI_INCLUDE_DIR})
  target_link_libraries(slopcap-camprobe PRIVATE ${ASI_LIBRARY})
  target_compile_definitions(slopcap-camprobe PRIVATE CAMPROBE_ASI)
  message(STATUS "ZWO ASI SDK: ${ASI_LIBRARY}")
else()
  message(STATUS "ZWO ASI SDK not found - ASI probe disabled")
endif()

# QHY SDK. Headers and library must come from the same install: this machine has the packaged SDK
# (/usr/include/libqhy + /usr/lib/x86_64-linux-gnu) and a newer manual one in /usr/local whose library the loader
# finds first. Default to the packaged one and pin it with RUNPATH; override with -DQHY_ROOT=/usr/local.
set(QHY_ROOT "" CACHE PATH "QHY SDK prefix (default: the distro package)")
if(QHY_ROOT)
  find_path(QHY_INCLUDE_DIR qhyccd.h PATHS ${QHY_ROOT}/include ${QHY_ROOT}/include/libqhy NO_DEFAULT_PATH)
  find_library(QHY_LIBRARY qhyccd PATHS ${QHY_ROOT}/lib NO_DEFAULT_PATH)
else()
  find_path(QHY_INCLUDE_DIR qhyccd.h PATHS /usr/include/libqhy NO_DEFAULT_PATH)
  find_library(QHY_LIBRARY qhyccd PATHS /usr/lib/x86_64-linux-gnu NO_DEFAULT_PATH)
endif()
if(QHY_INCLUDE_DIR AND QHY_LIBRARY)
  target_sources(slopcap-camprobe PRIVATE probe_qhy.cpp)
  target_include_directories(slopcap-camprobe PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/compat/qhy ${QHY_INCLUDE_DIR})
  target_link_libraries(slopcap-camprobe PRIVATE ${QHY_LIBRARY})
  target_compile_definitions(slopcap-camprobe PRIVATE CAMPROBE_QHY)
  get_filename_component(QHY_LIB_DIR ${QHY_LIBRARY} DIRECTORY)
  set_target_properties(slopcap-camprobe PROPERTIES BUILD_RPATH ${QHY_LIB_DIR} INSTALL_RPATH ${QHY_LIB_DIR})
  target_link_options(slopcap-camprobe PRIVATE -Wl,--enable-new-dtags)
  message(STATUS "QHY SDK: ${QHY_LIBRARY} (headers ${QHY_INCLUDE_DIR})")
else()
  message(STATUS "QHY SDK not found - QHY probe disabled")
endif()
